local function createTeam(name, color)
	if not Teams:FindFirstChild(name) then
		local newTeam = Instance.new("Team")
		newTeam.Name = name
		newTeam.TeamColor = color
		newTeam.AutoAssignable = false
		newTeam.Parent = Teams
	end
end

-- -----------------------------------------------------------------------------
-- 2. ADIM: LIDERSTATS, MAAS VE KAFADAKI ORGUT YAZISI
-- -----------------------------------------------------------------------------
local OrgutSistemiData = {}

Players.PlayerAdded:Connect(function(player)
	local leaderstats = Instance.new("Folder")
	leaderstats.Name = "leaderstats"
	leaderstats.Parent = player

	local akce = Instance.new("IntValue")
	akce.Name = "Akce"
	akce.Value = 1000
	akce.Parent = leaderstats

	local orgutVal = Instance.new("StringValue")
	orgutVal.Name = "Orgut"
	orgutVal.Value = "Yok"
	orgutVal.Parent = player
	
	player.CharacterAdded:Connect(function(char)
		local head = char:WaitForChild("Head")
		local billboard = Instance.new("BillboardGui")
		billboard.Size = UDim2.new(0, 200, 0, 50)
		billboard.StudsOffset = Vector3.new(0, 3, 0)
		billboard.Name = "OrgutYazisiTag"
		
		local textLabel = Instance.new("TextLabel")
		textLabel.Size = UDim2.new(1, 0, 1, 0)
		textLabel.BackgroundTransparency = 1
		textLabel.TextColor3 = Color3.fromRGB(255, 215, 0)
		textLabel.TextScaled = true
		textLabel.Font = Enum.Font.SourceSansBold
		textLabel.Text = "Örgüt: " .. orgutVal.Value
		textLabel.Parent = billboard
		billboard.Parent = head
		
		orgutVal.Changed:Connect(function(newVal)
			textLabel.Text = "Örgüt: " .. newVal
		end)
	end)

	-- 10 DAKIKADA BIR OTOMATIK 250 AKCE MAAŞ MOTORU
	task.spawn(function()
		while task.wait(600) do
			if player and player:Parent then
				akce.Value = akce.Value + 250
			end
		end
	end)
end)

-- -----------------------------------------------------------------------------
-- 3. ADIM: REMOTE EVENTS NETWORK (AG BAGLANTILARI)
-- -----------------------------------------------------------------------------
local HapisEvent = ReplicatedStorage:FindFirstChild("HapisEvent") or Instance.new("RemoteEvent")
HapisEvent.Name = "HapisEvent"
HapisEvent.Parent = ReplicatedStorage

local OrgutEvent = ReplicatedStorage:FindFirstChild("OrgutEvent") or Instance.new("RemoteEvent")
OrgutEvent.Name = "OrgutEvent"
OrgutEvent.Parent = ReplicatedStorage

HapisEvent.OnServerEvent:Connect(function(zaptiye, targetPlayerName, sure, sebep)
	if zaptiye.Team and zaptiye.Team.Name == "Zaptiye Amirligi" then
		local mahkum = Players:FindFirstChild(targetPlayerName)
		if mahkum and mahkum.Character then
			local eskiTakim = mahkum.Team
			mahkum.Team = Teams["Vatandaslar / Orgutler"]
			mahkum.Character:MoveTo(HapisSpawnPart.Position + Vector3.new(0, 3, 0))
			
			task.wait(tonumber(sure) or 10)
			
			if mahkum and mahkum:Parent then
				mahkum.Team = eskiTakim
				local anaSpawn = workspace:FindFirstChild("SpawnLocation")
				if anaSpawn and mahkum.Character then
					mahkum.Character:MoveTo(anaSpawn.Position)
				end
			end
		end
	end
end)

OrgutEvent.OnServerEvent:Connect(function(player, action, orgutName, orgutAmac)
	local akce = player.leaderstats.Akce
	if action == "Kur" and akce.Value >= 500 and player.Orgut.Value == "Yok" then
		akce.Value = akce.Value - 500
		player.Orgut.Value = orgutName
		OrgutSistemiData[orgutName] = {Lider = player.Name, Amac = orgutAmac, Uyeler = {player.Name}}
		OrgutEvent:FireAllClients("Guncelle", OrgutSistemiData)
	elseif action == "Katil" and OrgutSistemiData[orgutName] and player.Orgut.Value == "Yok" then
		player.Orgut.Value = orgutName
		table.insert(OrgutSistemiData[orgutName].Uyeler, player.Name)
		OrgutEvent:FireAllClients("Guncelle", OrgutSistemiData)
	end
end)
